   220→export function MaterialsView() {
   221→  const { user } = useAuth()
   222→  const isFaculty = user?.role === 'FACULTY'
   223→
   224→  const [materials, setMaterials] = useState<Material[]>([])
   225→  const [subjects, setSubjects] = useState<Subject[]>([])
   226→  const [classrooms, setClassrooms] = useState<ClassroomSummary[]>([])
   227→  const [loading, setLoading] = useState(true)
   228→  const [error, setError] = useState<string | null>(null)
   229→
   230→  const [selectedSubject, setSelectedSubject] = useState<string>(AUTO_SUBJECT)
   231→  const [selectedClassroom, setSelectedClassroom] = useState<string>(NO_CLASSROOM)
   232→  const [activeTab, setActiveTab] = useState<TabId>('all')
   233→  const [search, setSearch] = useState('')
   234→  const [dragOver, setDragOver] = useState(false)
   235→  const [uploadQueue, setUploadQueue] = useState<UploadJob[]>([])
   236→
   237→  const [dialogOpen, setDialogOpen] = useState(false)
   238→  const [newSubjectName, setNewSubjectName] = useState('')
   239→  const [creating, setCreating] = useState(false)
   240→
   241→  const fileInputRef = useRef<HTMLInputElement>(null)
   242→
   243→  const load = useCallback(async () => {
   244→    setLoading(true)
   245→    setError(null)
   246→    try {
   247→      const [mats, subs] = await Promise.all([
   248→        api.get<Material[]>('/materials'),
   249→        api.get<Subject[]>('/subjects'),
   250→      ])
   251→      setMaterials(mats)
   252→      setSubjects(subs)
   253→      if (isFaculty) {
   254→        try {
   255→          const rooms = await api.get<ClassroomSummary[]>('/classrooms')
   256→          setClassrooms(rooms)
   257→        } catch {
   258→          setClassrooms([]) // non-fatal — classroom targeting is optional
   259→        }
   260→      }
   261→    } catch (err) {
   262→      setError(err instanceof Error ? err.message : 'Failed to load materials')
   263→    } finally {
   264→      setLoading(false)
   265→    }
   266→  }, [isFaculty])
   267→
   268→  useEffect(() => {
   269→    load()
   270→  }, [load])
   271→
   272→  /* rotate the animated AI step label for every in-flight upload */
   273→  const anyUploading = uploadQueue.some(u => u.status === 'uploading')
   274→  useEffect(() => {
   275→    if (!anyUploading) return
   276→    const t = window.setInterval(() => {
   277→      setUploadQueue(q =>
   278→        q.map(u =>
   279→          u.status === 'uploading' ? { ...u, stage: (u.stage + 1) % AI_STEPS.length } : u
   280→        )
   281→      )
   282→    }, 4000)
   283→    return () => window.clearInterval(t)
   284→  }, [anyUploading])
   285→
   286→  /* ---------- upload (drag & drop + browse) ---------- */
   287→
   288→  const handleFiles = useCallback(
   289→    async (fileList: FileList | File[]) => {
   290→      const files = Array.from(fileList)
   291→      if (files.length === 0) return
   292→
   293→      const tooBig = files.filter(f => f.size > MAX_SIZE)
   294→      const valid = files.filter(f => f.size <= MAX_SIZE)
   295→      if (tooBig.length > 0) {
   296→        toast.error(
   297→          tooBig.length === 1
   298→            ? `"${tooBig[0]!.name}" exceeds the 10MB limit`
   299→            : `${tooBig.length} files exceed the 10MB limit`
   300→        )
   301→      }
   302→      if (valid.length === 0) return
   303→
   304→      const materialType = activeTab === 'all' ? 'NOTES' : UPLOAD_TYPE[activeTab]
   305→      const jobs: UploadJob[] = valid.map(f => ({
   306→        id: crypto.randomUUID(),
   307→        name: f.name,
   308→        status: 'uploading' as const,
   309→        stage: 0,
   310→      }))
   311→      setUploadQueue(q => [...q, ...jobs])
   312→
   313→      let ok = 0
   314→      for (let i = 0; i < valid.length; i++) {
   315→        const file = valid[i]!
   316→        const job = jobs[i]!
   317→        try {
   318→          const fd = new FormData()
   319→          fd.append('file', file)
   320→          fd.append('material_type', materialType)
   321→          if (selectedSubject !== AUTO_SUBJECT) fd.append('subject_id', selectedSubject)
   322→          if (isFaculty && selectedClassroom !== NO_CLASSROOM) {
   323→            fd.append('classroom_id', selectedClassroom)
   324→          }
   325→          const res = await api.upload<MaterialUploadAccepted>('/materials/upload', fd)
   326→          ok++
   327→          setUploadQueue(q => q.map(u => (u.id === job.id ? { ...u, status: 'done' } : u)))
   328→          toast.success(`"${file.name}" uploaded — AI is analyzing it in the background`)
   329→        } catch (err) {
   330→          setUploadQueue(q => q.map(u => (u.id === job.id ? { ...u, status: 'error' } : u)))
   331→          toast.error(`"${file.name}" failed: ${err instanceof Error ? err.message : 'upload error'}`)
   332→        }
   333→      }
   334→
   335→      if (ok > 0) {
   336→        try {
   337→          // subjects/materials both change when the AI pipeline runs
   338→          const [mats, subs] = await Promise.all([
   339→            api.get<Material[]>('/materials'),
   340→            api.get<Subject[]>('/subjects'),
   341→          ])
   342→          setMaterials(mats)
   343→          setSubjects(subs)
   344→        } catch {
   345→          /* keep current list on refresh failure */
   346→        }
   347→      }
   348→
   349→      // fade the queue out once uploads settle
   350→      window.setTimeout(() => {
   351→        setUploadQueue(q => q.filter(u => !jobs.some(j => j.id === u.id)))
   352→      }, 3500)
   353→    },
   354→    [selectedSubject, selectedClassroom, activeTab, isFaculty]
   355→  )
   356→
   357→  /* ---------- subject creation (manual, secondary) ---------- */
   358→
   359→  const createSubject = async () => {
   360→    const name = newSubjectName.trim()
   361→    if (!name) {
   362→      toast.error('Enter a subject name')
   363→      return
   364→    }
   365→    setCreating(true)
   366→    try {
   367→      const sub = await api.post<Subject>('/subjects', { name })
   368→      setSubjects(prev => [...prev, sub].sort((a, b) => a.name.localeCompare(b.name)))
   369→      setSelectedSubject(sub.id)
   370→      setDialogOpen(false)
   371→      setNewSubjectName('')
   372→      toast.success(`Subject "${sub.name}" created`)
   373→    } catch (err) {
   374→      toast.error(err instanceof Error ? err.message : 'Failed to create subject')
   375→    } finally {
   376→      setCreating(false)
   377→    }
   378→  }
   379→
   380→  /* ---------- derived ---------- */
   381→
   382→  const onboarding = materials.length === 0 && subjects.length === 0
   383→
   384→  const tabCount = (id: TabId) =>
   385→    id === 'all' ? materials.length : materials.filter(m => m.type === id).length
   386→
   387→  const filtered = materials
   388→    .filter(m => activeTab === 'all' || m.type === activeTab)
   389→    .filter(m => m.name.toLowerCase().includes(search.trim().toLowerCase()))
   390→
   391→  if (loading) return <div className="animate-fade-in-up"><MaterialsSkeleton /></div>
   392→  if (error) {
   393→    return (
   394→      <div className="animate-fade-in-up">
   395→        <ViewHeader title="Study Materials" subtitle="Upload and manage your study files" />
   396→        <ErrorCard message={error} onRetry={load} />
   397→      </div>
   398→    )
   399→  }
   400→
   401→  const uploading = anyUploading
   402→
   403→  return (
   404→    <div className="animate-fade-in-up space-y-6">
   405→      <ViewHeader
   406→        title="Study Materials"
   407→        subtitle="Drop a file and the AI builds the subject, question bank and your study plan"
   408→      >
   409→        <AIBadge label="AI Analyzed" />
   410→      </ViewHeader>
   411→
   412→      {/* onboarding hero — first run, nothing uploaded yet */}
   413→      {onboarding && <OnboardingHero />}
   414→
   415→      {/* Upload target row — hidden for students in onboarding (auto-detect is the flow) */}
   416→      {(!onboarding || isFaculty) && (
   417→        <div className="flex flex-col gap-3 sm:flex-row sm:items-end sm:gap-4">
   418→          <div className="w-full sm:w-64">
   419→            <Label htmlFor="upload-subject" className="text-xs font-medium text-slate-500">
   420→              Subject
   421→            </Label>
   422→            <Select value={selectedSubject} onValueChange={setSelectedSubject}>
   423→              <SelectTrigger id="upload-subject" className="mt-1.5 w-full bg-white">
   424→                <SelectValue placeholder="Choose a subject…" />
   425→              </SelectTrigger>
   426→              <SelectContent>
   427→                <SelectItem value={AUTO_SUBJECT} className="font-medium">
   428→                  ✨ Auto-detect from file (AI)
   429→                </SelectItem>
   430→                {subjects.map(s => (
   431→                  <SelectItem key={s.id} value={s.id}>
   432→                    {s.name}
   433→                  </SelectItem>
   434→                ))}
   435→              </SelectContent>
   436→            </Select>
   437→          </div>
   438→
   439→          {isFaculty && (
   440→            <div className="w-full sm:w-64">
   441→              <Label htmlFor="upload-classroom" className="text-xs font-medium text-slate-500">
   442→                Upload into classroom
   443→              </Label>
   444→              {classrooms.length > 0 ? (
   445→                <Select value={selectedClassroom} onValueChange={setSelectedClassroom}>
   446→                  <SelectTrigger id="upload-classroom" className="mt-1.5 w-full bg-white">
   447→                    <SelectValue placeholder="Choose a classroom…" />
   448→                  </SelectTrigger>
   449→                  <SelectContent>
   450→                    <SelectItem value={NO_CLASSROOM}>Not shared (personal)</SelectItem>
   451→                    {classrooms.map(c => (
   452→                      <SelectItem key={c.id} value={c.id}>
   453→                        {c.name}
   454→                      </SelectItem>
   455→                    ))}
   456→                  </SelectContent>
   457→                </Select>
   458→              ) : (
   459→                <p className="mt-1.5 flex h-9 items-center gap-1.5 text-xs text-slate-400">
   460→                  <Users className="h-3.5 w-3.5 shrink-0" />
   461→                  Create a classroom to share materials with students
   462→                </p>
   463→              )}
   464→            </div>
   465→          )}
   466→
   467→          <Button
   468→            variant="ghost"
   469→            size="sm"
   470→            className="h-9 shrink-0 text-xs text-slate-500 hover:text-violet-700"
   471→            onClick={() => setDialogOpen(true)}
   472→          >
   473→            <Plus className="mr-1 h-3.5 w-3.5" /> Add subject manually
   474→          </Button>
   475→          <p className="hidden text-xs text-slate-400 lg:block">
   476→            Auto-detect lets the AI pick the subject from the file content.
   477→          </p>
   478→        </div>
   479→      )}
   480→
   481→      {/* Drag & drop upload zone — the primary interaction */}
   482→      <div>
   483→        <div
   484→          role="button"
   485→          tabIndex={0}
   486→          aria-label="Upload study materials"
   487→          onClick={() => {
   488→            if (!uploading) fileInputRef.current?.click()
   489→          }}
   490→          onKeyDown={e => {
   491→            if (e.key === 'Enter' || e.key === ' ') {
   492→              e.preventDefault()
   493→              if (!uploading) fileInputRef.current?.click()
   494→            }
   495→          }}
   496→          onDragOver={e => {
   497→            e.preventDefault()
   498→            e.stopPropagation()
   499→            setDragOver(true)
   500→          }}
   501→          onDragLeave={e => {
   502→            e.preventDefault()
   503→            e.stopPropagation()
   504→            setDragOver(false)
   505→          }}
   506→          onDrop={e => {
   507→            e.preventDefault()
   508→            e.stopPropagation()
   509→            setDragOver(false)
   510→            void handleFiles(e.dataTransfer.files)
   511→          }}
   512→          className={cn(
   513→            'flex cursor-pointer flex-col items-center justify-center gap-2.5 rounded-xl border-2 border-dashed px-6 py-10 text-center outline-none transition-all focus-visible:ring-2 focus-visible:ring-violet-300',
   514→            dragOver
   515→              ? 'scale-[1.01] border-violet-500 bg-violet-50/70'
   516→              : 'border-slate-300 bg-white hover:border-violet-400 hover:bg-violet-50/40'
   517→          )}
   518→        >
   519→          <input
   520→            ref={fileInputRef}
   521→            type="file"
   522→            hidden
   523→            multiple
   524→            accept=".pdf,.doc,.docx,.jpg,.png"
   525→            onChange={e => {
   526→              if (e.target.files && e.target.files.length > 0) void handleFiles(e.target.files)
   527→              e.target.value = '' // allow re-picking the same file
   528→            }}
   529→          />
   530→          <div
   531→            className={cn(
   532→              'flex h-12 w-12 items-center justify-center rounded-full transition-colors',
   533→              dragOver ? 'bg-violet-100 text-violet-600' : 'bg-violet-50 text-violet-500'
   534→            )}
   535→          >
   536→            {uploading ? <Loader2 className="h-5 w-5 animate-spin" /> : <Upload className="h-5 w-5" />}
   537→          </div>
   538→          <h3 className="text-sm font-semibold text-slate-800">
   539→            {uploading ? 'AI is analyzing your files…' : 'Upload Study Materials'}
   540→          </h3>
   541→          <p className="text-xs text-slate-500">
   542→            Drag &amp; drop files or click to browse · PDF, DOC, Images · Max 10MB each
   543→          </p>
   544→        </div>
   545→        <p className="mt-2 text-center text-[11px] text-slate-400">
   546→          Uploads run the full AI pipeline — expect 20–60 seconds per file.
   547→        </p>
   548→
   549→        {/* per-file AI pipeline progress */}
   550→        {uploadQueue.length > 0 && (
   551→          <div className="mt-3 space-y-2 rounded-xl border border-violet-100 bg-violet-50/40 p-3">
   552→            {uploadQueue.map(job => (
   553→              <div key={job.id} className="flex items-center gap-2.5 text-sm">
   554→                {job.status === 'uploading' && (
   555→                  <Loader2 className="h-4 w-4 shrink-0 animate-spin text-violet-600" />
   556→                )}
   557→                {job.status === 'done' && (
   558→                  <CheckCircle2 className="h-4 w-4 shrink-0 text-emerald-600" />
   559→                )}
   560→                {job.status === 'error' && <XCircle className="h-4 w-4 shrink-0 text-red-500" />}
   561→                <span className="min-w-0 flex-1 truncate text-slate-700">{job.name}</span>
   562→                <span
   563→                  className={cn(
   564→                    'shrink-0 text-xs font-medium',
   565→                    job.status === 'uploading' && 'text-violet-600',
   566→                    job.status === 'done' && 'text-emerald-600',
   567→                    job.status === 'error' && 'text-red-600'
   568→                  )}
   569→                >
   570→                  {job.status === 'uploading'
   571→                    ? AI_STEPS[job.stage]
   572→                    : job.status === 'done'
   573→                      ? 'Done'
   574→                      : 'Failed'}
   575→                </span>
   576→              </div>
   577→            ))}
   578→          </div>
   579→        )}
   580→      </div>
   581→
   582→      {/* Tabs + search + grid — only meaningful once something exists */}
   583→      {!onboarding && (
   584→        <>
   585→          <div className="flex flex-col gap-3 lg:flex-row lg:items-center lg:justify-between">
   586→            <Tabs value={activeTab} onValueChange={v => setActiveTab(v as TabId)}>
   587→              <TabsList className="h-auto w-full flex-wrap justify-start gap-1 bg-slate-100 p-1 lg:w-auto">
   588→                {TABS.map(t => (
   589→                  <TabsTrigger
   590→                    key={t.id}
   591→                    value={t.id}
   592→                    className="rounded-md px-3 py-1.5 text-xs font-medium text-slate-600 data-[state=active]:bg-white data-[state=active]:text-violet-700 data-[state=active]:shadow-xs"
   593→                  >
   594→                    {t.label}
   595→                    <span className="ml-1.5 rounded-full bg-slate-200/80 px-1.5 py-0.5 text-[10px] font-semibold tabular-nums text-slate-600">
   596→                      {tabCount(t.id)}
   597→                    </span>
   598→                  </TabsTrigger>
   599→                ))}
   600→              </TabsList>
   601→            </Tabs>
   602→            <div className="relative w-full lg:w-64">
   603→              <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-slate-400" />
   604→              <Input
   605→                value={search}
   606→                onChange={e => setSearch(e.target.value)}
   607→                placeholder="Search materials…"
   608→                className="h-9 bg-white pl-9 text-sm"
   609→              />
   610→            </div>
   611→          </div>
   612→
   613→          {/* Materials grid */}
   614→          {filtered.length === 0 ? (
   615→            <EmptyState
   616→              icon={FolderOpen}
   617→              title={search.trim() ? 'No materials match your search' : 'No materials found'}
   618→              message={
   619→                search.trim()
   620→                  ? `Nothing matches "${search.trim()}" in this tab.`
   621→                  : activeTab === 'all'
   622→                    ? 'Upload your syllabus, PYQs, notes, or lab manuals to get started.'
   623→                    : `No ${TYPE_META[activeTab as MaterialType].label} yet — drop files above and they will show up here.`
   624→              }
   625→            />
   626→          ) : (
   627→            <div className="grid gap-4 [grid-template-columns:repeat(auto-fill,minmax(240px,1fr))]">
   628→              {filtered.map((mat, i) => {
   629→                const meta = TYPE_META[mat.type] ?? TYPE_META.notes
   630→                return (
   631→                  <Card
   632→                    key={mat.id}
   633→                    className={cn(
   634→                      'animate-fade-in-up shadow-xs transition-shadow hover:shadow-md',
   635→                      STAGGER[i % STAGGER.length]
   636→                    )}
   637→                  >
   638→                    <CardContent className="flex items-start gap-3.5 p-4">
   639→                      <div
   640→                        className={cn(
   641→                          'flex h-11 w-11 shrink-0 items-center justify-center rounded-xl',
   642→                          meta.tile
   643→                        )}
   644→                      >
   645→                        <meta.icon className="h-5 w-5" />
   646→                      </div>
   647→                      <div className="min-w-0 flex-1">
   648→                        <p title={mat.name} className="truncate text-sm font-semibold text-slate-800">
   649→                          {mat.name}
   650→                        </p>
   651→                        <p className="mt-0.5 truncate text-xs text-slate-400">
   652→                          {mat.subject} · {mat.size}
   653→                        </p>
   654→                        <div className="mt-2.5 flex flex-wrap items-center gap-1.5">
   655→                          {statusBadge(mat.status)}
   656→                          {mat.analysisSummary && (
   657→                            <Badge
   658→                              variant="outline"
   659→                              title={`${mat.analysisSummary.subjectName}: ${mat.analysisSummary.units} units · ${mat.analysisSummary.topics} topics · ${mat.analysisSummary.questions} questions`}
   660→                              className="border-emerald-200 bg-emerald-50 text-[11px] font-medium text-emerald-700"
   661→                            >
   662→                              🤖 {mat.analysisSummary.units}u · {mat.analysisSummary.topics}t ·{' '}
   663→                              {mat.analysisSummary.questions}q
   664→                            </Badge>
   665→                          )}
   666→                          {mat.uploadedByName && (
   667→                            <Badge
   668→                              variant="outline"
   669→                              title={`Uploaded by ${mat.uploadedByName}`}
   670→                              className="border-violet-200 bg-violet-100 text-[11px] font-medium text-violet-700"
   671→                            >
   672→                              by {mat.uploadedByName}
   673→                            </Badge>
   674→                          )}
   675→                        </div>
   676→                        <span className="mt-1.5 block text-[11px] text-slate-400">
   677→                          {formatDate(mat.uploadDate)}
   678→                        </span>
   679→                      </div>
   680→                    </CardContent>
   681→                  </Card>
   682→                )
   683→              })}
   684→            </div>
   685→          )}
   686→        </>
   687→      )}
   688→
   689→      {/* Add subject dialog (manual, rarely needed) */}
   690→      <Dialog open={dialogOpen} onOpenChange={setDialogOpen}>
   691→        <DialogContent className="sm:max-w-sm">
   692→          <DialogHeader>
   693→            <DialogTitle>Add Subject Manually</DialogTitle>
   694→            <DialogDescription>
   695→              Usually unnecessary — the AI auto-detects subjects from your uploads. Use this only
   696→              for a subject with no files yet.
   697→            </DialogDescription>
   698→          </DialogHeader>
   699→          <form
   700→            className="space-y-4"
   701→            onSubmit={e => {
   702→              e.preventDefault()
   703→              void createSubject()
   704→            }}
   705→          >
   706→            <div className="space-y-2">
   707→              <Label htmlFor="new-subject">Subject name</Label>
   708→              <Input
   709→                id="new-subject"
   710→                autoFocus
   711→                value={newSubjectName}
   712→                onChange={e => setNewSubjectName(e.target.value)}
   713→                placeholder="e.g. Machine Learning"
   714→                className="h-10"
   715→                maxLength={80}
   716→              />
   717→            </div>
   718→            <DialogFooter>
   719→              <Button
   720→                type="button"
   721→                variant="outline"
   722→                onClick={() => setDialogOpen(false)}
   723→                disabled={creating}
   724→              >
   725→                Cancel
   726→              </Button>
   727→              <Button type="submit" disabled={creating} className="bg-violet-600 hover:bg-violet-700">
   728→                {creating && <Loader2 className="mr-2 h-4 w-4 animate-spin" />}
   729→                {creating ? 'Creating…' : 'Create Subject'}
   730→              </Button>
   731→            </DialogFooter>
   732→          </form>
   733→        </DialogContent>
   734→      </Dialog>
   735→    </div>
   736→  )
   737→}
   738→
