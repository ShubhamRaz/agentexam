   103→export function TheoryExamView({ onNavigate }: { onNavigate: (v: ViewId) => void }) {
   104→  const [phase, setPhase] = useState<Phase>('setup')
   105→  const [subjects, setSubjects] = useState<Subject[]>([])
   106→  const [subjectsLoading, setSubjectsLoading] = useState(true)
   107→  const [subjectsError, setSubjectsError] = useState<string | null>(null)
   108→
   109→  const [config, setConfig] = useState<ExamConfig | null>(null)
   110→  const [exam, setExam] = useState<ExamStartResponse | null>(null)
   111→  const [totalMarks, setTotalMarks] = useState(0)
   112→  const [result, setResult] = useState<ExamResult | null>(null)
   113→  const [creating, setCreating] = useState(false)
   114→
   115→  const loadSubjects = useCallback(() => {
   116→    setSubjectsLoading(true)
   117→    setSubjectsError(null)
   118→    api
   119→      .get<Subject[]>('/subjects')
   120→      .then(data => setSubjects(data || []))
   121→      .catch(err => setSubjectsError(err instanceof Error ? err.message : 'Failed to load subjects'))
   122→      .finally(() => setSubjectsLoading(false))
   123→  }, [])
   124→
   125→  useEffect(() => {
   126→    loadSubjects()
   127→  }, [loadSubjects])
   128→
   129→  /* create + start the exam, then move to the exam phase */
   130→  const handleStart = useCallback(async (cfg: ExamConfig) => {
   131→    setCreating(true)
   132→    try {
   133→      const created = await api.post<{ id: string; total_marks: number; question_count: number }>(
   134→        '/exams',
   135→        {
   136→          subject_id: cfg.subjectId,
   137→          difficulty: cfg.difficulty,
   138→          question_count: cfg.questionCount,
   139→          duration_minutes: cfg.durationMinutes,
   140→        }
   141→      )
   142→      const started = await api.post<ExamStartResponse>(`/exams/${created.id}/start`)
   143→      setConfig(cfg)
   144→      setExam(started)
   145→      setTotalMarks(created.total_marks)
   146→      setPhase('exam')
   147→    } catch (err) {
   148→      toast.error(err instanceof Error ? err.message : 'Failed to start the exam. Please try again.')
   149→    } finally {
   150→      setCreating(false)
   151→    }
   152→  }, [])
   153→
   154→  const handleExamFinished = useCallback((res: ExamResult) => {
   155→    setResult(res)
   156→    setPhase('results')
   157→  }, [])
   158→
   159→  const restart = useCallback(() => {
   160→    setResult(null)
   161→    setExam(null)
   162→    setTotalMarks(0)
   163→    setPhase('setup')
   164→  }, [])
   165→
   166→  /* ---------- exam phase ---------- */
   167→  if (phase === 'exam' && exam && config) {
   168→    return (
   169→      <div className="animate-fade-in-up space-y-6">
   170→        <ActiveExam exam={exam} config={config} totalMarks={totalMarks} onFinished={handleExamFinished} />
   171→      </div>
   172→    )
   173→  }
   174→
   175→  /* ---------- results phase ---------- */
   176→  if (phase === 'results' && result && config) {
   177→    return (
   178→      <div className="animate-fade-in-up space-y-6">
   179→        <ViewHeader title="Exam Results" subtitle={`${config.subjectName} — AI-evaluated mock test`} />
   180→        <ExamResults
   181→          result={result}
   182→          questions={exam?.questions ?? []}
   183→          onRestart={restart}
   184→          onPerformance={() => onNavigate('performance')}
   185→        />
   186→      </div>
   187→    )
   188→  }
   189→
   190→  /* ---------- setup phase (default) ---------- */
   191→  return (
   192→    <div className="animate-fade-in-up space-y-6">
   193→      <ViewHeader
   194→        title="Theory Exam"
   195→        subtitle="AI-graded mock tests — objective answers scored instantly, written answers evaluated by your AI examiner"
   196→      />
   197→      {subjectsError ? (
   198→        <ErrorCard message={subjectsError} onRetry={loadSubjects} />
   199→      ) : (
   200→        <ExamSetup
   201→          subjects={subjects}
   202→          subjectsLoading={subjectsLoading}
   203→          creating={creating}
   204→          onStart={handleStart}
   205→        />
   206→      )}
   207→    </div>
   208→  )
   209→}

   342→
   343→function ActiveExam({
   344→  exam,
   345→  config,
   346→  totalMarks,
   347→  onFinished,
   348→}: {
   349→  exam: ExamStartResponse
   350→  config: ExamConfig
   351→  totalMarks: number
   352→  onFinished: (result: ExamResult) => void
   353→}) {
   354→  const questions = exam.questions
   355→
   356→  /* answers: questionId → selected option id (MCQ) or answer text */
   357→  const [answers, setAnswers] = useState<Record<string, string>>(() => {
   358→    const initial: Record<string, string> = {}
   359→    for (const saved of exam.saved_answers) {
   360→      const value = saved.selected_option_id ?? saved.answer_text ?? ''
   361→      if (value) initial[saved.question_id] = value
   362→    }
   363→    return initial
   364→  })
   365→  const [marked, setMarked] = useState<Set<string>>(new Set())
   366→  const [current, setCurrent] = useState(0)
   367→  const [secondsLeft, setSecondsLeft] = useState(config.durationMinutes * 60)
   368→  const [showConfirm, setShowConfirm] = useState(false)
   369→  const [submitting, setSubmitting] = useState(false)
   370→  const [evalStep, setEvalStep] = useState(0)
   371→
   372→  /* refs to dodge stale closures in timers + submit guard */
   373→  const submittingRef = useRef(false)
   374→  const timeUpRef = useRef(false)
   375→  const pendingRef = useRef<{ qid: string; text: string } | null>(null)
   376→  const debounceRef = useRef<number | null>(null)
   377→  const stepTimersRef = useRef<number[]>([])
   378→  const submitRef = useRef<() => void>(() => {})
   379→
   380→  const question = questions[current]
   381→  const total = questions.length
   382→
   383→  const isAnswered = useCallback(
   384→    (q: ExamQuestion) => {
   385→      const a = answers[q.id]
   386→      if (a === undefined) return false
   387→      return q.question_type === 'MCQ' ? a !== '' : a.trim() !== ''
   388→    },
   389→    [answers]
   390→  )
   391→  const answeredCount = questions.filter(isAnswered).length
   392→  const unanswered = total - answeredCount
   393→
   394→  /* ---------- autosave (fire-and-forget) ---------- */
   395→
   396→  const flushPending = useCallback(async () => {
   397→    if (debounceRef.current !== null) {
   398→      window.clearTimeout(debounceRef.current)
   399→      debounceRef.current = null
   400→    }
   401→    const pending = pendingRef.current
   402→    pendingRef.current = null
   403→    if (pending) {
   404→      try {
   405→        await api.put(`/exams/${exam.id}/answers/${pending.qid}`, { answer_text: pending.text })
   406→      } catch (err) {
   407→        console.error('Failed to save answer', err)
   408→      }
   409→    }
   410→  }, [exam.id])
   411→
   412→  const handleTextChange = (qid: string, value: string) => {
   413→    setAnswers(prev => ({ ...prev, [qid]: value }))
   414→    pendingRef.current = { qid, text: value }
   415→    if (debounceRef.current !== null) window.clearTimeout(debounceRef.current)
   416→    debounceRef.current = window.setTimeout(() => {
   417→      const pending = pendingRef.current
   418→      pendingRef.current = null
   419→      debounceRef.current = null
   420→      if (pending) {
   421→        api
   422→          .put(`/exams/${exam.id}/answers/${pending.qid}`, { answer_text: pending.text })
   423→          .catch(err => console.error('Failed to save answer', err))
   424→      }
   425→    }, DEBOUNCE_MS)
   426→  }
   427→
   428→  const selectOption = (qid: string, optionId: string) => {
   429→    setAnswers(prev => ({ ...prev, [qid]: optionId }))
   430→    api
   431→      .put(`/exams/${exam.id}/answers/${qid}`, { selected_option_id: optionId })
   432→      .catch(err => console.error('Failed to save answer', err))
   433→  }
   434→
   435→  /* flush any pending debounced text when navigating between questions or unmounting */
   436→  useEffect(() => {
   437→    return () => {
   438→      void flushPending()
   439→    }
   440→  }, [current, flushPending])
   441→
   442→  /* ---------- submit (manual or on timer expiry) ---------- */
   443→
   444→  const submitExam = useCallback(async () => {
   445→    if (submittingRef.current) return
   446→    submittingRef.current = true
   447→    setSubmitting(true)
   448→    setShowConfirm(false)
   449→    setEvalStep(0)
   450→    stepTimersRef.current = [
   451→      window.setTimeout(() => setEvalStep(1), 2600),
   452→      window.setTimeout(() => setEvalStep(2), 9000),
   453→    ]
   454→    try {
   455→      await flushPending()
   456→      const res = await api.post<ExamResult>(`/exams/${exam.id}/submit`)
   457→      stepTimersRef.current.forEach(t => window.clearTimeout(t))
   458→      stepTimersRef.current = []
   459→      setEvalStep(EVAL_STEPS.length)
   460→      await new Promise(r => window.setTimeout(r, 900))
   461→      onFinished(res)
   462→    } catch (err) {
   463→      stepTimersRef.current.forEach(t => window.clearTimeout(t))
   464→      stepTimersRef.current = []
   465→      submittingRef.current = false
   466→      setSubmitting(false)
   467→      setEvalStep(0)
   468→      toast.error(err instanceof Error ? err.message : 'Failed to submit the exam. Please try again.')
   469→    }
   470→  }, [exam.id, flushPending, onFinished])
   471→
   472→  useEffect(() => {
   473→    submitRef.current = () => void submitExam()
   474→  }, [submitExam])
   475→
   476→  /* countdown — paused while the AI evaluation overlay is up */
   477→  useEffect(() => {
   478→    if (submitting) return
   479→    const t = window.setInterval(() => {
   480→      setSecondsLeft(s => Math.max(0, s - 1))
   481→    }, 1000)
   482→    return () => window.clearInterval(t)
   483→  }, [submitting])
   484→
   485→  /* auto-submit on expiry */
   486→  useEffect(() => {
   487→    if (secondsLeft === 0 && !timeUpRef.current) {
   488→      timeUpRef.current = true
   489→      toast.info("Time's up — submitting your test for evaluation.")
   490→      submitRef.current()
   491→    }
   492→  }, [secondsLeft])
   493→
   494→  /* cleanup pending timers on unmount */
   495→  useEffect(() => {
   496→    return () => {
   497→      stepTimersRef.current.forEach(t => window.clearTimeout(t))
   498→      if (debounceRef.current !== null) window.clearTimeout(debounceRef.current)
   499→    }
   500→  }, [])
   501→
   502→  const toggleMark = (qid: string) => {
   503→    setMarked(prev => {
   504→      const next = new Set(prev)
   505→      if (next.has(qid)) next.delete(qid)
   506→      else next.add(qid)
   507→      return next
   508→    })
   509→  }
   510→
   511→  if (!question || total === 0) {
   512→    return (
   513→      <div className="animate-fade-in-up space-y-6">
   514→        <ViewHeader title="Theory Exam" subtitle="No questions available for this configuration." />
   515→        <ErrorCard message="This paper has no questions. Go back and try another subject or difficulty." />
   516→      </div>
   517→    )
   518→  }
   519→
   520→  const timerDanger = secondsLeft < 60
   521→  const timerWarn = !timerDanger && secondsLeft < 300
   522→
   523→  return (
   524→    <div className="space-y-6">
   525→      {/* sticky exam header — acts as this phase's view header */}
   526→      <div className="sticky top-0 z-30 rounded-xl border border-slate-200 bg-white/95 px-4 py-3 shadow-xs backdrop-blur">
   527→        <div className="flex flex-wrap items-center justify-between gap-3">
   528→          <div className="min-w-0">
   529→            <h2 className="truncate text-base font-bold text-slate-900 sm:text-lg">
   530→              {config.subjectName} — Mock Test
   531→            </h2>
   532→            <div className="mt-1 flex flex-wrap items-center gap-1.5">
   533→              <Badge variant="outline" className="text-[11px] font-medium text-slate-600">
   534→                Question {current + 1} of {total}
   535→              </Badge>
   536→              {totalMarks > 0 && (
   537→                <Badge variant="outline" className="text-[11px] font-medium text-slate-600">
   538→                  {totalMarks} marks
   539→                </Badge>
   540→              )}
   541→            </div>
   542→          </div>
   543→          <div className="flex items-center gap-2.5">
   544→            <div
   545→              className={cn(
   546→                'flex items-center gap-1.5 rounded-full border px-3 py-1.5 text-sm font-bold tabular-nums',
   547→                timerDanger && 'timer-danger border-red-200 bg-red-50 text-red-600',
   548→                timerWarn && 'border-amber-200 bg-amber-50 text-amber-700',
   549→                !timerDanger && !timerWarn && 'border-slate-200 bg-white text-slate-700'
   550→              )}
   551→              aria-live="polite"
   552→            >
   553→              <Clock className="h-4 w-4" />
   554→              {formatClock(secondsLeft)}
   555→            </div>
   556→            <Button
   557→              size="sm"
   558→              className="bg-red-600 text-white hover:bg-red-700"
   559→              disabled={submitting}
   560→              onClick={() => setShowConfirm(true)}
   561→            >
   562→              <Send className="h-3.5 w-3.5" /> Submit Test
   563→            </Button>
   564→          </div>
   565→        </div>
   566→      </div>
   567→
   568→      <div className="flex flex-col gap-6 lg:flex-row">
   569→        {/* main column */}
   570→        <div className="min-w-0 flex-1 space-y-6">
   571→          {/* compact progress for mobile (palette is desktop-only) */}
   572→          <div className="lg:hidden">
   573→            <div className="flex items-center justify-between text-xs text-slate-500">
   574→              <span>
   575→                Question {current + 1} of {total}
   576→                {marked.size > 0 && <span className="ml-2 text-amber-600">· {marked.size} marked</span>}
   577→              </span>
   578→              <span className="font-semibold text-violet-600">

   867→
   868→  const optionLabel = (qid: string, optionId: string | null): string => {
   869→    if (!optionId) return '—'

   876→    if (!optionId) return '—'

   881→    switch (grade) {
   882→      case 'A':
   883→        return 'border-emerald-200 bg-emerald-50 text-emerald-700'
   884→      case 'B':
   885→        return 'border-cyan-200 bg-cyan-50 text-cyan-700'
   886→      case 'C':
   887→        return 'border-amber-200 bg-amber-50 text-amber-700'
   888→      case 'D':
   889→        return 'border-orange-200 bg-orange-50 text-orange-700'

   908→              {result.obtained_marks} / {result.total_marks} marks
   909→            </Badge>
   910→            <Badge variant="outline" className={cn('px-3 py-1 text-sm font-bold', gradeStyle(result.grade))}>
   911→              Grade {result.grade}
   912→            </Badge>
   913→          </div>
   914→        </CardContent>
   915→      </Card>

   919→        <AICard title="AI Feedback" icon={Sparkles}>
   920→          <p className="whitespace-pre-line">{result.analytics.improvement_suggestions}</p>
   921→          {result.analytics.weak_areas.length > 0 && (
   922→            <div className="mt-4 flex flex-wrap items-center gap-1.5">
   923→              <span className="text-xs font-semibold text-slate-500">Focus areas:</span>

   925→                <Badge
   926→                  key={area}
   927→                  variant="outline"
   928→                  className="border-amber-200 bg-amber-50 text-[11px] font-medium text-amber-700"
   929→                >
   930→                  {area}
   931→                </Badge>
   932→              ))}
   933→            </div>
   934→          )}
   935→        </AICard>
   936→      )}
   937→

   939→      <Card className="shadow-xs">
   940→        <div className="border-b border-slate-100 px-5 py-4">
   941→          <h3 className="text-sm font-semibold text-slate-900">Question Breakdown</h3>

   943→            Your answers vs the expected answers, with per-question AI feedback.

   949→            const border =
   950→              ratio >= 0.999
