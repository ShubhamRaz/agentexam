    25→  options?: Opt[]
    26→  correct?: string
    27→  explanation?: string
    28→  marks?: number
    29→  difficulty?: 'EASY' | 'MEDIUM' | 'HARD'
    30→  source?: 'PYQ' | 'MANUAL' | 'AI'
    31→  year?: number
    32→  sourceRef?: string
    33→}

    39→  name: string

    48→const mcq = (text: string, topic: string, opts: string[], correctIdx: number, explanation: string, difficulty: 'EASY' | 'MEDIUM' | 'HARD' = 'MEDIUM'): SeedQ => ({
    49→  type: 'MCQ', text, topic,
    50→  options: opts.map((t, i) => ({ id: String.fromCharCode(65 + i), text: t, is_correct: i === correctIdx })),
    51→  correct: String.fromCharCode(65 + correctIdx),
    52→  explanation, marks: 1, difficulty, source: 'AI',
    53→})
    54→
    55→const short = (text: string, topic: string, marks = 3, difficulty: 'EASY' | 'MEDIUM' | 'HARD' = 'MEDIUM'): SeedQ => ({

    57→})

    59→const long = (text: string, topic: string, marks = 8, difficulty: 'EASY' | 'MEDIUM' | 'HARD' = 'HARD'): SeedQ => ({

    61→})

    72→    correct: String.fromCharCode(65 + correctIdx!),

    74→  } : {}),
    75→})
