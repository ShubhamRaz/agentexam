    91→/** DELETE via fetch — mirrors api-client (which has no delete wrapper). */
    92→async function apiDelete(path: string): Promise<void> {
    93→  const headers: Record<string, string> = {}
    94→  const token = getToken()
    95→  if (token) headers['Authorization'] = `Bearer ${token}`
    96→  const res = await fetch(`/api${path}`, { method: 'DELETE', headers })
    97→  if (!res.ok) {
    98→    let message = `Request failed (${res.status})`
    99→    try {
   100→      const data = (await res.json()) as { error?: string; message?: string }
   101→      message = data.error || data.message || message
   102→    } catch { /* ignore parse errors */ }
   103→    throw new ApiError(message, res.status)
   104→  }
   105→}

   153→/** amber within 3 days, red when overdue */
   154→function dueDateClass(due: string): string {
   155→  const daysLeft = (new Date(due).getTime() - Date.now()) / 86_400_000
   156→  if (daysLeft < 0) return 'font-semibold text-red-600'
   157→  if (daysLeft <= 3) return 'font-semibold text-amber-600'
   158→  return 'text-slate-500'
   159→}

   192→function AddStudentDialog({ open, onOpenChange, classroomId, onAdded }: {
   193→  open: boolean
   194→  onOpenChange: (open: boolean) => void
   195→  classroomId: string
   196→  onAdded: () => void
   197→}) {
   198→  const [email, setEmail] = useState('')
   199→  const [name, setName] = useState('')
   200→  const [adding, setAdding] = useState(false)
   201→
   202→  const submit = async () => {
   203→    const trimmedEmail = email.trim()
   204→    if (!trimmedEmail) {
   205→      toast.error('Enter a student email')
   206→      return
   207→    }
   208→    setAdding(true)
   209→    try {
   210→      const res = await api.post<{ student: ClassroomStudentRow; tempPassword: string | null }>(
   211→        `/classrooms/${classroomId}/students`,
   212→        { email: trimmedEmail, name: name.trim() || undefined }
   213→      )
   214→      toast.success(`Student added — ${res.student.name}`, {
   215→        description: res.tempPassword ? `Temporary password: ${res.tempPassword}` : undefined,
   216→      })
   217→      setEmail('')
   218→      setName('')
   219→      onOpenChange(false)
   220→      onAdded()
   221→    } catch (err) {
   222→      toast.error(err instanceof Error ? err.message : 'Failed to add student')
   223→    } finally {
   224→      setAdding(false)
   225→    }
   226→  }
   227→
   228→  return (
   229→    <Dialog open={open} onOpenChange={onOpenChange}>
   230→      <DialogContent className="sm:max-w-sm">
   231→        <DialogHeader>
   232→          <DialogTitle>Add Student</DialogTitle>
   233→          <DialogDescription>
   234→            Enroll an existing student or invite a new one by email.
   235→          </DialogDescription>
   236→        </DialogHeader>
   237→        <form
   238→          className="space-y-4"
   239→          onSubmit={e => {
   240→            e.preventDefault()
   241→            void submit()
   242→          }}
   243→        >
   244→          <div className="space-y-2">
   245→            <Label htmlFor="student-email">Student email</Label>
   246→            <Input
   247→              id="student-email"
   248→              type="email"
   249→              autoFocus
   250→              value={email}
   251→              onChange={e => setEmail(e.target.value)}
   252→              placeholder="student@college.edu"
   253→              className="h-10"
   254→            />
   255→          </div>
   256→          <div className="space-y-2">
   257→            <Label htmlFor="student-name">
   258→              Name <span className="font-normal text-slate-400">(optional)</span>
   259→            </Label>
   260→            <Input
   261→              id="student-name"
   262→              value={name}
   263→              onChange={e => setName(e.target.value)}
   264→              placeholder="Derived from the email when left blank"
   265→              className="h-10"
   266→              maxLength={80}
   267→            />
   268→          </div>
   269→          <p className="rounded-lg bg-violet-50/70 px-3 py-2.5 text-xs leading-relaxed text-violet-700">
   270→            <UserPlus className="mr-1 inline h-3.5 w-3.5" />
   271→            New students get a temporary password (<code className="font-semibold">student123</code>)
   272→            — share it with them so they can log in.

   274→          <DialogFooter>
   275→            <Button type="button" variant="outline" onClick={() => onOpenChange(false)} disabled={adding}>

   676→            size="icon"
   677→            className="h-9 w-9 shrink-0"
   678→            onClick={onBack}
   679→            aria-label="Back to faculty dashboard"
   680→          >
   681→            <ArrowLeft className="h-4 w-4" />
   682→          </Button>

   686→              <code className="rounded-md border border-slate-200 bg-slate-50 px-1.5 py-0.5 text-[11px] font-semibold tracking-wider text-slate-600">
   687→                {detail.code}
