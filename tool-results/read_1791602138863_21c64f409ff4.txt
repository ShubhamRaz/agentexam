    36→const mcq = (text: string, topic: string, opts: string[], correctIdx: number, explanation: string, difficulty: 'EASY' | 'MEDIUM' | 'HARD' = 'MEDIUM'): SeedQ => ({
    37→  type: 'MCQ', text, topic,
    38→  options: opts.map((t, i) => ({ id: String.fromCharCode(65 + i), text: t, is_correct: i === correctIdx })),
    39→  correct: String.fromCharCode(65 + correctIdx),
    40→  explanation, marks: 1, difficulty, source: 'MANUAL',
    41→})
    42→
    43→const short = (text: string, topic: string, marks = 5, difficulty: 'EASY' | 'MEDIUM' | 'HARD' = 'MEDIUM'): SeedQ => ({

    45→})
    46→
    47→const long = (text: string, topic: string, marks = 10, difficulty: 'EASY' | 'MEDIUM' | 'HARD' = 'HARD'): SeedQ => ({
    48→  type: 'LONG_ANSWER', text, topic, marks, difficulty, source: 'MANUAL',
    49→})
    50→
    51→const viva = (text: string, topic: string, marks = 2): SeedQ => ({
    52→  type: 'VIVA', text, topic, marks, difficulty: 'MEDIUM', source: 'MANUAL',
    53→})

    55→const pyq = (text: string, topic: string, year: number, marks: number, type: 'MCQ' | 'SHORT_ANSWER' | 'LONG_ANSWER', difficulty: 'EASY' | 'MEDIUM' | 'HARD' = 'MEDIUM', opts?: string[], correctIdx?: number): SeedQ => ({
